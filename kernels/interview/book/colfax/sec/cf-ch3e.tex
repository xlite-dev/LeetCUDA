\subsection{嵌套元组态射上的运算}
我们的下一个任务是发展一套「嵌套元组态射（nested tuple morphism）代数」。由于我们已经为元组态射建立过这样一套「代数」，我们可以通过为各种运算的输出配备适当的轮廓（profile），把这套代数推广到嵌套情形。



\subsubsection{拼接}

接下来，我们定义嵌套元组态射上的一个拼接（concatenate）运算，它与布局（layout）的拼接相容，即
\[
L_{(f,g)} = (L_f,L_g).
\]
我们通过拼接 $f$ 与 $g$ 的定义域（domain）来拼接嵌套元组态射 $f$ 与 $g$。为使这一定义良定义（well-defined），需要 $f$ 与 $g$ 满足一个不交性条件，其确切表述见下文。

\begin{definition}\label{definitionofdisjointimagesinD} 设 $f$ 与 $g$ 是具有相同陪域（codomain）的嵌套元组态射。若 $f^\flat$ 与 $g^\flat$ 如定义 \ref{definitionofdisjointimagesinC} 中那样具有不交的像，则称 $f$ 与 $g$ 具有{\it 不交的像}（disjoint images）。
\end{definition}

\begin{example}
\label{exampleofdisjointimagesinD1}
    若
    \[
    f:(3,(512,512)) \to (2,512,2,512)
    \]
    位于（lies over）$(*,2,4)$ 之上，且
    \[
    g:(2,2) \to (2,512,2,512)
    \]
    位于 $(1,3)$ 之上，则 $f$ 与 $g$ 具有不交的像。
\end{example}

\begin{example}
\label{exampleofdisjointimagesinD2}
若
\[f : (2,(32,64)) \to (32,(2,2,2),64)\]
位于 $\alpha = (3,1,5)$ 之上，且
\[g:((2,2)) \to (32,(2,2,2),64)\]
位于 $\beta = (2,4)$ 之上，则 $f$ 与 $g$ 具有不交的像。
\end{example}


\begin{construction}\label{concatenationofmorphismsinD}
    设 $f:S \to T$ 与 $g:U \to T$ 是分别位于 $\alpha$ 与 $\beta$ 之上的嵌套元组态射，且 $f$ 与 $g$ 具有不交的像。我们把 $f$ 与 $g$ 的{\it 拼接}定义为嵌套元组态射
    \[
    (f,g) : (S,U) \to T
    \]
    它满足
    \begin{align*}
        \flatten((f,g)) & = f^\flat \star g^\flat.
    \end{align*}
    更一般地，若 $f_i:S_i \to T$（$1 \leq i \leq k$）是嵌套元组态射，且 $f_1,\dots,f_k$ 两两具有不交的像，则我们把{\it 拼接}
    \[
    (f_1,\dots,f_k) : (S_1 , \dots , S_k) \to T.
    \]
    定义为满足
    \begin{align*}
        (f_1, \dots , f_k)^\flat & = f_1^\flat \star \cdots \star f_k^\flat.
    \end{align*}

\end{construction}

\begin{example} 例 \ref{exampleofdisjointimagesinD1} 中态射 $f$ 与 $g$ 的拼接是嵌套元组态射
\[(f,g): \left((3,(512,512)),(2,2)\right) \to (2,512,2,512)\]
它位于 $\alpha \star \beta = (*,2,4,1,3)$ 之上。
\end{example}

\begin{example} 例 \ref{exampleofdisjointimagesinD2} 中态射 $f$ 与 $g$ 的拼接是嵌套元组态射
\[(f,g): ( (2,(32,64)),((2,2))) \to (32,(2,2,2),64)\]
它位于 $\alpha \star \beta = (3,1,5,2,4)$ 之上。
\end{example}

\begin{example}
    若
    \[
    f:(2,2) \to (2,3,5,2,3,5)
    \]
    位于 $\alpha = (1,4)$ 之上，
    \[
    g:(3,3) \to (2,3,5,2,3,5)
    \]
    位于 $\beta = (2,5)$ 之上，且
    \[
    h: (5,5) \to (2,3,5,2,3,5)
    \]
    位于 $\gamma = (3,6)$ 之上，则 $f$、$g$ 与 $h$ 两两具有不交的像，且拼接
    \[
    (f,g,h):((2,2),(3,3),(5,5)) \to (2,3,5,2,3,5)
    \]
    位于 $\alpha \star \beta \star \gamma = (1,4,2,5,3,6)$ 之上。
\end{example}

\begin{example}
设 $f:S \to T$ 是嵌套元组态射，并设
\[
S^\flat = (s_1,\dots,s_m).
\]
回顾例 \ref{definitionofentriesofnestedtuplemorphisms}：对任意 $1 \leq i \leq m$，存在嵌套元组态射
\[
f_i:s_i \to T.
\]
称为 $f$ 的第 $i$ 个元素（entry）。这些态射两两具有不交的像，而拼接
\[
(f_1,\dots,f_m):S^\flat \to T
\]
是例 \ref{definitionofflatteningisomorphisms} 中的复合（composite）
\[
(f_1,\dots,f_m) = f \circ \id_{S^\flat}^S
\]
\end{example}

\begin{example}
设 $f:S \to T$ 是嵌套元组态射，并设
\[
S = (S_1,\dots,S_r).
\]
回顾例 \ref{definitionofmodesofnestedtuplemorphisms}：对任意 $1 \leq i \leq r$，存在嵌套元组态射
\[
f_i:S_i \to T.
\]
称为 $f$ 的第 $i$ 个模式（mode）。这些态射两两具有不交的像，且拼接
\[
(f_1,\dots,f_r):S \to T
\]
等于 $f$。换言之，每个嵌套元组态射 $f$ 都可以写成它的各模式的拼接：
\[f = (f_1,\dots,f_r).\]
\end{example}




\begin{proposition}\label{compatibilityofconcatenateinD}
    若 $f_1,\dots,f_k$ 是具有相同陪域且两两具有不交像的嵌套元组态射，则
    \[
    L_{(f_1,\dots,f_k)} = (L_{f_1},\dots,L_{f_k}).
    \]
\end{proposition}

\begin{proof}
由构造（by construction）可知，我们有
\begin{align*}
 \shape((L_{f_1},\dots,L_{f_k})) & = (\shape(L_{f_1}),\dots,\shape(L_{f_k}))\\
& = \shape(L_{(f_1,\dots,f_k)}).
\end{align*}
再利用命题 \ref{concatenateinCproposition}，可得
\begin{align*}
(L_{f_1},\dots,L_{f_k})^\flat & = L_{f_1}^\flat \star \cdots \star L_{f_k}^\flat\\
& = L_{f_1^\flat} \star \cdots \star L_{f_k^\flat} \\
& = L_{f_1^\flat \star \cdots \star f_k^\flat }\\
& = L_{(f_1,\dots, f_k)^\flat}\\
& = (L_{(f_1,\dots,f_k)})^\flat.
\end{align*}
\end{proof}


\subsubsection{合并}

若 $f$ 是嵌套元组态射，我们本可以把合并（coalesce）$\coalesce(f)$ 定义为 $\coalesce^\flat(f^\flat)$。从理论上讲，这是一个健全的定义。然而，为了让我们的定义与 \texttt{cute} 实现相容，我们要对 $\coalesce(f)$ 的定义做一点小修改。

\begin{definition}
\label{definitionofcoalesceofnestedtuplemorphism}
    设 $f:S \to T$ 是嵌套元组态射，并记
    \[
    \coalesce^\flat(f^\flat):(s_1,\dots,s_m) \to (t_1,\dots,t_n).
    \]
    \begin{itemize}
    \item （情形 1）：若 $m >1$，我们定义
    \[\coalesce(f) = \coalesce^\flat(f^\flat).\]
    \item （情形 2）：若 $m = 1$，我们把 $\coalesce(f)$ 定义为复合
    \[
    \begin{tikzcd}
    s_1 \ar[r,swap,"(1)"] &  (s_1) \ar[rr,"\coalesce^\flat(f^\flat)"] & & (t_1,\dots,t_n).
    \end{tikzcd}
    \]
    \item （情形 3）：若 $m = 0$，我们把 $\coalesce(f)$ 定义为复合
    \[
    \begin{tikzcd}
    1 \ar[r,swap,"(*)"] &  () \ar[rr,"\coalesce^\flat(f^\flat)"] & & (t_1,\dots,t_n).
    \end{tikzcd}
    \]
    \end{itemize}
\end{definition}

\begin{example}
    若
    \[f:((2,2),(3,3),(5,5)) \to (5,5,3,3,2,2)\]
    位于 $\alpha = (5,6,3,4,1,2)$ 之上，则
    \[
    \coalesce(f):(4,9,25) \to (25,9,4)
    \]
    位于 $\alpha' = (3,2,1)$ 之上。
\end{example}

\begin{proposition}\label{coalesceagreement}
    若 $f:S \to T$ 是嵌套元组态射，则
    \[
    \coalesce(L_f) = L_{\coalesce(f)}.
    \]
\end{proposition}

\begin{proof}
我们再次写出
    \[ \begin{tikzcd}
    (s_1,\dots,s_m) \ar[rr,"\coalesce^\flat(f^\flat)"] \ar[rr,swap,"\alpha"] & & (t_1,\dots,t_n).
    \end{tikzcd} \]
    需考虑三种情形。
    \begin{itemize}
        \item （情形 1）：设 $m > 1$。则
        \begin{align*}
        L_{\coalesce(f)} & = L_{\coalesce^\flat(f^\flat)}\\
        & = \coalesce^\flat(L_{f^\flat})\\
        & = \coalesce((L_f)^\flat)\\
        & = \coalesce(L_f).
    \end{align*}
    \item （情形 2）：设 $m = 1$。则
    \begin{align*}
        L_{\coalesce(f)} & = s_1:t_1 \dots,t_{\alpha(1)-1}\\
        & = \coalesce((s_1):(t_1\cdots
        t_{\alpha(1)-1}))\\
        &= \coalesce(L_{\coalesce^\flat(f^\flat)})\\
        & = \coalesce(\coal^\flat(L_{f^\flat}))\\
        & = \coalesce((L_f)^\flat)\\
        & = \coalesce(L_f).
    \end{align*}
    \item （情形 3）：设 $m = 0$。则
    \begin{align*}
        L_{\coalesce(f)} & = 1:0 \\
        & = \coalesce(():())\\
        &= \coalesce(L_{\coalesce^\flat(f^\flat)})\\
        & = \coalesce(\coal^\flat(L_{f^\flat}))\\
        & = \coalesce((L_f)^\flat)\\
        & = \coalesce(L_f).
    \end{align*}
    \end{itemize}

\end{proof}

\subsubsection{求补}

本节中，我们定义嵌套元组态射互补的概念。

\begin{definition} \label{definitionofcomplementarymorphismsinD}
设 $f:S \to T$ 与 $g:U \to T$ 是具有不交像的嵌套元组态射。若
\[
(f,g):(S,U) \to T
\]
是一个同构（isomorphism），则称 $g$ 为 $f$ 的一个{\it 补}（complement）。
\end{definition}

\begin{remark}若 $f:S \to T$ 与 $g:U \to T$ 是嵌套元组态射，则 $g$ 是 $f$ 的补当且仅当 $g^\flat$ 是 $f^\flat$ 的补，因为 $(f,g)^\flat = f^\flat \star g^\flat$。
\end{remark}




\begin{proposition}\label{complementofmorphisminDproposition}
    若 $f:S \to T$ 是嵌套元组态射，且 $g:U \to T$ 是 $f$ 的补，则 $L_{g}$ 是 $L_{f}$ 的 $\size(T)$-补。
\end{proposition}

\begin{proof}
    观察 \ref{flatlayoutcompatibility} 蕴含
    \begin{align*} (L_{f})^\flat & = L_{f^\flat}\text{, and }\\
    (L_{g})^\flat & = L_{g^\flat}
    \end{align*}
    而引理 \ref{flatteningcomplementationlemma} 使我们可以化归到扁平情形（命题 \ref{complementofmorphisminCproposition}）。
\end{proof}


\begin{construction}\label{complementofmorphisminD}
    设 $f:S \to T$ 是嵌套元组态射。我们把 $f$ 的{\it 补}定义为复合
    \[ \begin{tikzcd}
    U \ar[dr,swap,"(f^\flat)^c"] \ar[rr,"f^c"] & & T \\
    & T^\flat \ar[ur,swap,"\id_{T^\flat}^T"]
    \end{tikzcd} \]
    其中 $(f^\flat)^c$ 如构造 \ref{complementofmorphisminC} 中所定义，而 $\id_{T^\flat}^T:T^\flat \cong T$ 是去扁平化同构（unflattening isomorphism）。
\end{construction}

\begin{example}
    嵌套元组态射
    \[\begin{tikzcd} ((2,2),(5,5)) \ar[rr,"f"] \ar[rr,swap,"(1{,}4{,}2{,}5)"]& &  ((2,5,7),(2,5,7))\end{tikzcd} \]
    的补是
    \[\begin{tikzcd} (7,7) \ar[rr,"f^c"] \ar[rr,swap,"(3{,}6)"]& &  ((2,5,7),(2,5,7)).\end{tikzcd} \]
\end{example}

\begin{proposition}
    设 $f:S \to T$ 与 $g:U \to T$ 是嵌套元组态射。若 $f$ 是单射且 $g$ 是 $f$ 的补，则 $L_{g}$ 是 $L_f$ 的 $\size(T)$-补。
\end{proposition}
\begin{proof}
    这由命题 \ref{complementofmorphisminCproposition} 与引理 \ref{flatteningcomplementationlemma} 即得，因为
    \begin{align*}
    (L_f)^\flat & = L_{f^\flat}\\
    (L_{g})^\flat & = L_{g^\flat}.
    \end{align*}
\end{proof}


\begin{proposition}\label{coalescedlayoutoftuplemorphismcomplement}
    若 $f:S \to T$ 是单射嵌套元组态射，则
    \[
    \coalesce(L_{f^c}) = \comp(L_f,\size(T)).
    \]
\end{proposition}
\begin{proof} 由于 $f^c$ 是由 $(f^\flat)^c$ 通过与一个重新加括号同构作后复合（post-compose）而得到的，于是
\[
L_{f^c} = L_{(f^\flat)^c}
\]
故由命题 \ref{coalescedlayoutoftuplemorphismcomplement} 可得
\[
\coal^\flat(L_{f^c})  = \comp^\flat(L_f,\size(T)).
\]
对两边应用 $\coal(-)$ 即得结论。
\end{proof}


\subsubsection{复合}

我们可以利用第 \ref{realizationofnestedtuplemorphisms} 节的实现函子（realization functor）来证明嵌套元组态射的复合（composition）与相应布局的复合相容。
\begin{theorem}\label{compatibilityofcompositioninD}
    若 $f$ 与 $g$ 是非退化（non-degenerate）且可复合的嵌套元组态射，则
    \[
    L_{g\circ f} = L_{g} \circ L_{f}.
    \]
\end{theorem}

\begin{proof}
    设 $f:S \to T$ 与 $g: T \to U$ 是非退化嵌套元组态射。我们需要验证
    \begin{enumerate}
        \item {\it $\shape(L_{g \circ f})$ 加细（refine）$\shape(L_{f})$}：这成立，因为
        \[\shape(L_f) = S = \shape(L_{g \circ f}).\]
        \item {\it $L_{g \circ f}$ 在 $\shape(L_{f})$ 上是已合并（coalesced）的}：这成立，因为嵌套元组态射 $g \circ f$ 是非退化的，从而布局 $L_{g \circ f}$ 也是非退化的。
        \item {\it $\Phi_{L_{g \circ f}} = \Phi_{L_{g}} \circ \Phi_{L_{f}}^{\size(L_{g})}$}：利用引理 \ref{nestedlayoutfunctionlemma}，我们有
        \begin{align*}
        \Phi_{L_{g \circ f}}^{\size(U)} &  = |g \circ f| \\
        &  = |g| \circ |f| \\
        & = \Phi_{L_g}^{\size(U)} \circ \Phi_{L_f}^{\size(T)}\\
        \end{align*}
        再与包含映射（inclusion）$[0,\size(U)) \subset \mathbb{Z}$ 作后复合，并注意到 $\size(T) = \size(L_{g})$，即得结论。
    \end{enumerate}
\end{proof}







\subsubsection{逻辑切分}

接下来，我们引入嵌套元组态射的逻辑切分（logical division）。这一构造（construction）由扁平切分（flat division）通过引入嵌套轮廓而得到，不带任何相容性约束。


\begin{definition}
     设 $f$ 与 $g$ 是嵌套元组态射。若 $g$ 与 $f$ 可复合，则称 $g$ {\it 整除}（divides）$f$。换言之，
     \[
     \codomain(g) = \domain(f).
     \]
\end{definition}

\begin{definition}\label{definitionoflogicaldivisionofnestedtuplemorphisms}
    设 $g:S \to T$ 与 $f:T \to U$ 是嵌套元组态射。我们把 $f$ 除以 $g$ 的逻辑切分定义为嵌套元组态射
    \[
    f \oslash g = f \circ (g,g^c).
    \]
\end{definition}

\begin{example}
    将
    \[\begin{tikzcd} ((2,2),2) \ar[rr,"f"] \ar[rr,swap,"(2{,}4{,}*)
"] & &  ((4,2),(4,2)) \end{tikzcd} \]
    除以
    \[ \begin{tikzcd} (2,2) \ar[rr,"g"] \ar[rr,swap,"(1{,}3)"] & &  ((2,2),2) \end{tikzcd} \]
    所得的逻辑切分是
    \[ \begin{tikzcd} ((2,2),2) \ar[rr,"f \oslash g"] \ar[rr,swap,"(2{,}*{,}4)"] & &  ((4,2),(4,2)). \end{tikzcd} \]
\end{example}

\begin{example}
    将
    \[\begin{tikzcd} (8,8,512,512,512) \ar[rr,"f"] \ar[rr,swap,"(*{,}*{,}1{,}2{,}3)"] & &  (512,512,512) \end{tikzcd} \]
    除以
    \[ \begin{tikzcd} (8,512) \ar[rr,"g"] \ar[rr,swap,"(1{,}5)"] & &  (8,8,512,512,512) \end{tikzcd} \]
    所得的逻辑切分是
    \[ \begin{tikzcd} ((8,512),(8,512,512)) \ar[rr,"f \oslash g"] \ar[rr,swap,"(*{,}1{,}*{,}2{,}3)"] & &  ((4,2),(4,2)). \end{tikzcd} \]
\end{example}



\begin{proposition}\label{coalesceoflogicaldivision}
    若 $g:S \to T$ 与 $f:T \to U$ 是非退化嵌套元组态射，则
    \[
    \coalesce(L_{f \oslash g}) = \coalesce(L_f \oslash L_g).
    \]
\end{proposition}

\begin{proof}
由命题 \ref{coalescedlayoutoftuplemorphismcomplement}，我们有
\[
\coalesce(\comp(L_g,\size(L_f))) = \coalesce(L_{g^c})
\]
并且我们计算
    \begin{align*}
        \coalesce(L_f \oslash L_g) & = \coalesce(L_f \circ \left(L_g , \comp(L_g,\size(L_f))\right))\\
        & = \coalesce( L_{f} \circ (L_g, L_{g^c}))\\
        & = \coalesce(L_{f} \circ L_{(g,g^c)})\\
        & = \coalesce(L_{f} \circ L_{(g,g^c)})\\
        & = \coalesce(L_{f \circ (g,g^c)})\\
        & = \coalesce(L_{f \oslash g}).
    \end{align*}
\end{proof}

\begin{proposition}
    若 $f$ 与 $g$ 是嵌套元组（nested tuple）且 $g$ 整除 $f$，则
    \[
    (f \oslash g)^\flat = f^\flat\oslash^\flat g^\flat.
    \]
\end{proposition}

\begin{proof}
    我们计算
    \begin{align*}
    (f \oslash g)^\flat & = (f \circ (g,g^c))^\flat\\
    & = f^\flat \circ (g,g^c)^\flat\\
    & = f^\flat \circ (g^\flat \star (g^c)^\flat)\\
    & = f^\flat \circ (g^\flat \star (g^\flat)^c)\\
    & = f^\flat \oslash^\flat g^\flat.
    \end{align*}
\end{proof}


\subsubsection{逻辑乘积}
本节中，我们定义嵌套元组态射的逻辑乘积。
\begin{definition}\label{definitionoflogicalproductofnestedtuplemorphisms}
设 $f$ 与 $g$ 是嵌套元组态射。若 $\codomain(g) = \domain(f^c)$，则称 $f$ 与 $g$ 是{\it 乘积可许的}（product admissible）。若 $f$ 与 $g$ 乘积可许，我们把 $f$ 与 $g$ 的{\it 逻辑乘积}（logical product）定义为嵌套元组态射
\[
f \otimes g = (f , f^c \circ g).
\]
\end{definition}




\begin{example}
    嵌套元组态射
    \[\begin{tikzcd}
    (8,8) \ar[rr,"f"] \ar[rr,swap,"(1{,}2)"] & &  (8,8,16,16)
    \end{tikzcd} \]
    与
    \[\begin{tikzcd}
    (16,16) \ar[rr,"g"] \ar[rr,swap,"(1{,}2)"] & &  (16,16)
    \end{tikzcd} \]
    是乘积可许的，它们的逻辑乘积是
    \[\begin{tikzcd}
    ((8,8),(16,16))  \ar[rr,"f\otimes g"] \ar[rr,swap,"(1{,}2{,}3{,}4)"] & &  (8,8,16,16).
    \end{tikzcd} \]
\end{example}
\begin{example}
    嵌套元组态射
    \[\begin{tikzcd}
    (128,128) \ar[rr,"f"] \ar[rr,swap,"(3{,}4)"] & &  (32,32,128,128)
    \end{tikzcd} \]
    与
    \[\begin{tikzcd}
    (32) \ar[rr,"g"] \ar[rr,swap,"(2)"] & &  (32,32)
    \end{tikzcd} \]
    是乘积可许的，它们的逻辑乘积是
    \[\begin{tikzcd}
    ((128,128),(32))  \ar[rr,"f\otimes g"] \ar[rr,swap,"(3{,}4{,}2)"] & &  (32,32,128,128).
    \end{tikzcd} \]
\end{example}




\begin{proposition}\label{logicalproductcompatibility}
    设 $f$ 与 $g$ 是非退化嵌套元组态射且 $f$ 与 $g$ 乘积可许。则
    \[
    L_{f \otimes g} = L_f \otimes L_g.
    \]
\end{proposition}

\begin{proof} 设 $f:S \to T$ 与 $g:U \to V$ 乘积可许，并令
    \[
    L_f^* = \comp(L_f,\size(L_f)\cdot \cosize(L_g))
    \]
    由于 $f$ 是单射且 $\codomain(g) = \domain(f^c)$，可知
    \[\size(L_f) \cdot \cosize(L_g) \leq \size(S) \cdot \size(V) = \size(T).\]
    利用这一事实，以及事实
    \[\Phi_{\comp(L_f,\size(T))} = \Phi_{L_{f^c}},\]
    我们有
    \begin{align*}
    L_f^* \circ L_g & = \comp(L_f,\size(T)) \circ L_g\\
    & = L_{f^c} \circ L_g.
    \end{align*}
    再利用这一事实，我们计算
    \begin{align*}
        L_f \otimes L_g & = (L_f , L_f^* \circ L_g)\\
        & = (L_f , L_{f^c} \circ L_g)\\
        & = (L_f , L_{f^c \circ g})\\
        & = L_{(f , f^c \circ g)}\\
        & = L_{f \otimes g}
    \end{align*}
\end{proof}
